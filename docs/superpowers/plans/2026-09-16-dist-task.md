# 分发通知 → 本机任务表 → 接受安装 实现计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.
> 本仓库约定：**默认 inline 执行**（AGENTS.md 强制），不要用 subagent 驱动。

**Goal:** 群主分发技能/工作流后广播一条 `kind:'dist'` 群消息（target=all，含类型与下载地址），成员本机按类型建任务，点「接受」后把 skills 装到本机 agent 技能目录、workflow 与本地融合（同名覆盖）。

**Architecture:** 通知走现有群聊通道（新消息 kind + meta 结构化字段）；任务落单一 `void.db` 的新表 `task` / `task_cursor`（模块 `server/task-store.js`，与 `seed-store.js` 同构）；纯逻辑抽到 `server/collaboration/dist-task.js` 与 `server/collaboration/workflow-merge.js` 便于单测；安装复用既有 `installBundleZip`。

**Tech Stack:** Node 24 ESM（`node:sqlite`、`node --test`）、Vue3 + TS + Element Plus（vitest，纯逻辑测试，无 @vue/test-utils）。

**Spec:** `docs/superpowers/specs/2026-09-16-dist-task-design.md`

```js
import fs from 'node:fs';
import path from 'node:path';
import { DatabaseSync } from 'node:sqlite';

// 本机任务表：分发通知（kind:'dist'）在成员侧落成的待办；游标表记录「已扫到哪条群消息」
// 与 seed-store.js 同构：同一个 <数据目录>/void.db，自己建自己的表
export function createTaskStore({ file, logger = console } = {}) {
  fs.mkdirSync(path.dirname(file), { recursive: true });
  const db = new DatabaseSync(file);
  db.exec(`
    CREATE TABLE IF NOT EXISTS task (
      id TEXT PRIMARY KEY,
      group_id TEXT NOT NULL,
      hub_id TEXT,
      kind TEXT NOT NULL,
      hash TEXT NOT NULL,
      name TEXT,
      url TEXT,
      payload TEXT,
      status TEXT NOT NULL,
      message_id TEXT,
      created_at INTEGER,
      updated_at INTEGER,
      UNIQUE (group_id, kind, hash)
    );
    CREATE TABLE IF NOT EXISTS task_cursor (
      group_id TEXT PRIMARY KEY,
      last_ts INTEGER,
      last_id TEXT,
      updated_at INTEGER
    );
  `);

  // 冲突时只刷新展示/来源字段：status 原地保留（done / ignored / failed 不会被拉回 pending）
  const upsertStmt = db.prepare(`INSERT INTO task
    (id, group_id, hub_id, kind, hash, name, url, payload, status, message_id, created_at, updated_at)
    VALUES (?,?,?,?,?,?,?,?,?,?,?,?)
    ON CONFLICT(group_id, kind, hash) DO UPDATE SET
      hub_id=excluded.hub_id, name=excluded.name, url=excluded.url, payload=excluded.payload,
      message_id=excluded.message_id, updated_at=excluded.updated_at`);
  const findStmt = db.prepare('SELECT * FROM task WHERE group_id = ? AND kind = ? AND hash = ?');
  const getStmt = db.prepare('SELECT * FROM task WHERE id = ?');
  const listStmt = db.prepare('SELECT * FROM task');
  const setStmt = db.prepare('UPDATE task SET status = ?, payload = COALESCE(?, payload), updated_at = ? WHERE id = ?');
  const getCursorStmt = db.prepare('SELECT * FROM task_cursor WHERE group_id = ?');
  const setCursorStmt = db.prepare(`INSERT INTO task_cursor (group_id, last_ts, last_id, updated_at)
    VALUES (?,?,?,?) ON CONFLICT(group_id) DO UPDATE SET last_ts=excluded.last_ts, last_id=excluded.last_id, updated_at=excluded.updated_at`);

  const row = (r) => (r ? {
    id: r.id, groupId: r.group_id, hubId: r.hub_id, kind: r.kind, hash: r.hash,
    name: r.name, url: r.url, payload: (() => { try { return JSON.parse(r.payload || '{}'); } catch { return {}; } })(),
    status: r.status, messageId: r.message_id, createdAt: r.created_at, updatedAt: r.updated_at,
  } : null);
  const cursorRow = (r) => (r ? { groupId: r.group_id, lastTs: r.last_ts || 0, lastId: r.last_id || '', updatedAt: r.updated_at } : null);

  return {
    // 幂等：同 (group,kind,hash) 更新原行；返回 { task, created }
    upsert(rec) {
      const key = [String(rec.groupId), String(rec.kind), String(rec.hash)];
      const before = findStmt.get(...key);
      const id = before ? before.id : String(rec.id);
      try {
        upsertStmt.run(
          id, key[0], rec.hubId || null, key[1], key[2], rec.name || null, rec.url || null,
          JSON.stringify(rec.payload || {}), rec.status || 'pending', rec.messageId || null,
          Number(rec.createdAt) || Date.now(), Number(rec.updatedAt) || Date.now(),
        );
      } catch (e) { logger?.warn?.('task upsert failed', (e && e.message) || e); }
      return { task: row(getStmt.get(id)), created: !before };
    },
    get(id) { return row(getStmt.get(String(id))); },
    list({ status = null, kind = null, groupId = null } = {}) {
      return listStmt.all().map(row)
        .filter((t) => (!status || t.status === status) && (!kind || t.kind === kind) && (!groupId || t.groupId === groupId))
        .sort((a, b) => b.updatedAt - a.updatedAt);
    },
    setStatus(id, status, patch = {}) {
      try {
        setStmt.run(String(status), patch.payload ? JSON.stringify(patch.payload) : null, Date.now(), String(id));
      } catch (e) { logger?.warn?.('task setStatus failed', (e && e.message) || e); }
      return row(getStmt.get(String(id)));
    },
    getCursor(groupId) { return cursorRow(getCursorStmt.get(String(groupId))); },
    advanceCursor(groupId, ts, id) {
      const cur = cursorRow(getCursorStmt.get(String(groupId)));
      const next = Number(ts) || 0;
      if (cur && (cur.lastTs > next || (cur.lastTs === next && cur.lastId === String(id || '')))) return cur;
      try { setCursorStmt.run(String(groupId), next, String(id || ''), Date.now()); } catch { /* ignore */ }
      return cursorRow(getCursorStmt.get(String(groupId)));
    },
    close() { try { db.close(); } catch { /* ignore */ } },
    db,
  };
}
```

```js
// 分发通知（聊天消息 kind:'dist'）的纯逻辑：任务拆分、类型文案、下载地址、游标比较
// 抽成纯函数便于单测；engine.js 只做编排与 IO

// meta 里哪些资产非空 → 要建哪几类任务
export function distKindsOf(meta = {}) {
  const m = meta || {};
  const out = [];
  const hasSkills = ['skills', 'command', 'agent'].some((k) => Array.isArray(m[k]) && m[k].length > 0);
  if (hasSkills) out.push('skills');
  if (Array.isArray(m.workflow) && m.workflow.length > 0) out.push('workflow');
  return out;
}

// 仅用于正文/卡片文案：单类给单类，两类给 mixed，空兜底 skills
export function distTypeOf(meta = {}) {
  const kinds = distKindsOf(meta);
  if (kinds.length >= 2) return 'mixed';
  return kinds[0] || 'skills';
}

// 群主本机共享下载路由；name 含 '/' 时整段编码成单段（服务端按 '/' 切分后再 decodeURIComponent）
export function distUrl(host, port, name) {
  return `http://${host}:${port}/api/collab/downloads/file/${encodeURIComponent(String(name || ''))}`;
}

// 游标过滤：只增不减，同 ts 用 id 去重（重复扫描无副作用，游标只是省事）
export function shouldProcess(cursor, msg) {
  const ts = Number((msg && msg.ts) || 0);
  const lastTs = Number((cursor && cursor.lastTs) || 0);
  if (ts !== lastTs) return ts > lastTs;
  return String((msg && msg.id) || '') !== String((cursor && cursor.lastId) || '');
}
```

```js
// 工作流列表融合：分发/安装时把远端工作流并进本地
// 规则（spec §9）：label 同名 → 用远端内容覆盖、保留本地 id；未命中 → 按 id upsert
export function mergeWorkflowList(current = [], incoming = []) {
  const out = [];
  const byLabel = new Map();
  const byId = new Map();
  for (const w of Array.isArray(current) ? current : []) {
    if (!w || !w.id) continue;
    out.push({ ...w });
    byId.set(String(w.id), out[out.length - 1]);
    const key = String(w.label || '').trim();
    if (key) byLabel.set(key, out[out.length - 1]);
  }
  for (const raw of Array.isArray(incoming) ? incoming : []) {
    if (!raw || !raw.id || !Array.isArray(raw.stages)) continue;
    const key = String(raw.label || raw.id).trim();
    const hit = byLabel.get(key) || byId.get(String(raw.id));
    if (hit) {
      const keepId = hit.id;
      Object.assign(hit, { ...raw, id: keepId });
      byId.set(String(keepId), hit);
      if (key) byLabel.set(key, hit);
      continue;
    }
    const added = { ...raw };
    out.push(added);
    byId.set(String(added.id), added);
    if (key) byLabel.set(key, added);
  }
  return out;
}
```

```js
  async acceptTask(id) {
    const task = this.taskStore?.get?.(id);
    if (!task) return { code: 'not-found' };
    if (task.status !== 'pending' && task.status !== 'failed') return { code: 'not-pending' };
    if (!task.name || task.name.includes('..') || task.name.includes('\\')) {
      this.taskStore.setStatus(id, 'failed', { payload: { ...task.payload, lastError: 'bad-name' } });
      return { code: 'bad-name' };

    const abs = path.join(this.downloadsDir, task.name);
    if (!fs.existsSync(abs)) {
      try { await this.pullDownloads([task.name], task.hubId); } catch (e) { this.log?.warn?.('task pull failed', (e && e.message) || e); }
    }
    if (!fs.existsSync(abs)) {

      return { code: 'file-missing' };
    }
```

```ts
export interface DistTaskPayload {
  skills?: string[];
  command?: string[];
  agent?: string[];
  workflow?: { id: string; label: string }[];
  count?: number;
  size?: number;
  skillsUpdatedAt?: number;
  workflowUpdatedAt?: number;
  lastError?: string;

export interface DistTask {
  id: string;
  groupId: string;
  hubId?: string;
  kind: string;
  hash: string;
  name?: string;
  url?: string;
  status: string;
  messageId?: string;

export interface DistTaskGroups {
  pending: DistTask[];
  done: DistTask[];
  ignored: DistTask[];
  failed: DistTask[];
}

export function groupTasksByStatus(list: DistTask[]): DistTaskGroups {
  const out: DistTaskGroups = { pending: [], done: [], ignored: [], failed: [] };
  for (const t of list || []) {
    const bucket = (out as unknown as Record<string, DistTask[]>)[t.status];
    if (bucket) bucket.push(t);
  }
  return out;
}

export function kindKeyOf(kind: string): string {
  return kind === 'workflow' ? 'collab.task.workflow' : 'collab.task.skills';

  const n = (p.skills?.length || 0) + (p.command?.length || 0) + (p.agent?.length || 0) + (p.workflow?.length || 0);

}

export interface DistCard { type: string; name: string; url: string; count: number }

export function distCardOf(msg: { kind?: string; meta?: Record<string, unknown> } | null | undefined): DistCard | null {
  if (!msg || msg.kind !== 'dist' || !msg.meta) return null;
  const meta = msg.meta as { type?: string; name?: string; url?: string; count?: number };
  return { type: meta.type || 'skills', name: meta.name || '', url: meta.url || '', count: Number(meta.count) || 0 };
```
