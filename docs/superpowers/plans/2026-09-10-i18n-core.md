# i18n 核心基础设施（里程碑 1）实施计划

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 建立共享轻量 i18n 运行时与前后端接线（语言开关、持久化、服务端全局语言、守卫脚本），并以 `App.vue` 外壳文案迁移端到端验证整条链路。

**Architecture:** 词条与运行时集中在 `shared/i18n/`（纯函数，web/server 复用）；前端用 `web/src/i18n.ts` 包一层 Vue 响应式状态并注册全局 `$t`；服务端用 `server/i18n.js` 持有全局 locale（存 `config/server.json`）。默认语言 `zh-CN`，保证零行为回归。

**Tech Stack:** Node 24 ESM（`node:test`）、Vue 3.5 + TS + Vite、vitest。

**参考 spec:** `docs/superpowers/specs/2026-09-10-i18n-design.md`

**环境提醒:** Windows PowerShell 5.1，无 bash，一律 `npm.cmd`。

**里程碑边界:** 本计划只做「核心 + App.vue 外壳迁移」。Web 其余组件/模块的迁移（里程碑 2）与服务端用户可见文案迁移（里程碑 3）另出计划，均复用本计划产出的 `t()`/`$t()`/守卫。

```js
import { messages as zhCN } from './zh-CN.js';
import { messages as en } from './en.js';

export const LOCALES = Object.freeze(['zh-CN', 'en']);
export const DEFAULT_LOCALE = 'zh-CN';

const CATALOGS = { 'zh-CN': zhCN, en };

export function normalizeLocale(x) {
  if (typeof x !== 'string') return DEFAULT_LOCALE;
  const s = x.toLowerCase();
  if (s.startsWith('zh')) return 'zh-CN';
  if (s.startsWith('en')) return 'en';
  return DEFAULT_LOCALE;
}

function interpolate(str, params) {
  if (!params) return str;
  return str.replace(/\{(\w+)\}/g, (m, k) => (params[k] == null ? m : String(params[k])));
}

export function translate(locale, key, params) {
  const loc = normalizeLocale(locale);
  const current = CATALOGS[loc] || {};
  const raw = (key in current) ? current[key] : (key in en ? en[key] : key);
  return interpolate(raw, params);
}
```

- [ ] **Step 3: 创建 `web/src/i18n.ts`**

```ts
import { ref, type Ref } from 'vue';
import { translate, normalizeLocale, DEFAULT_LOCALE, LOCALES } from '@devflow/shared/i18n';

const STORE = 'devflow-lang';

function initial(): string {
  try {
    const saved = localStorage.getItem(STORE);
    if (saved) return normalizeLocale(saved);
    if (typeof navigator !== 'undefined' && navigator.language) return normalizeLocale(navigator.language);
  } catch { /* ignore */ }
  return DEFAULT_LOCALE;
}

export const locale: Ref<string> = ref(initial());

export function t(key: string, params?: Record<string, unknown>): string {
  return translate(locale.value, key, params);
}

export function setLocale(x: string) {
  const next = normalizeLocale(x);
  locale.value = next;
  try { localStorage.setItem(STORE, next); } catch { /* ignore */ }
  try {
    fetch('/api/settings/language', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ language: next }),
    }).catch(() => { /* 同步失败不影响本地切换 */ });
  } catch { /* ignore */ }
}

export function useI18n() {
  return { locale, t, setLocale, LOCALES };
}

declare module 'vue' {
  interface ComponentCustomProperties {
    $t: (key: string, params?: Record<string, unknown>) => string;
  }
}
```

```js
import fs from 'node:fs';
import path from 'node:path';
import { LOCALES, DEFAULT_LOCALE } from '@devflow/shared/i18n';

// 可配置的服务设置：config/server.json（端口、语言等）。修改端口需重启后生效。
export function defaultSettingsFile(baseDir) {
  return path.join(baseDir, 'config', 'server.json');
}
export function readSettings(file) {
  try {
    if (fs.existsSync(file)) {
      const j = JSON.parse(fs.readFileSync(file, 'utf8'));
      const language = LOCALES.includes(j.language) ? j.language : DEFAULT_LOCALE;
      return { port: Number(j.port) || null, language };
    }
  } catch { /* ignore */ }
  return { port: null, language: DEFAULT_LOCALE };
}
export function writeSettings(file, patch) {
  const cur = readSettings(file);
  const next = { ...cur, ...patch };
  fs.mkdirSync(path.dirname(file), { recursive: true });
  const tmp = file + '.tmp';
  fs.writeFileSync(tmp, JSON.stringify(next, null, 2), 'utf8');
  fs.renameSync(tmp, file);
  return next;
}
```

const SKIP_DIRS = new Set(['node_modules', 'dist', 'dist-portable', 'env', '.git']);
const SKIP_FILE = /\.test\.|\.d\.ts$/;

function walk(dir, out = []) {
  for (const e of fs.readdirSync(dir, { withFileTypes: true })) {
    if (e.name.startsWith('.') || SKIP_DIRS.has(e.name)) continue;
    const full = path.join(dir, e.name);
    if (e.isDirectory()) walk(full, out);
    else if (/\.(js|ts|vue)$/.test(e.name) && !SKIP_FILE.test(e.name)) out.push(full);
  }
  return out;
}
// 去掉块注释与行注释（避免把 http:// 当注释）
function stripComments(src) {
  let s = src.replace(/<!--[\s\S]*?-->/g, '');
  s = s.replace(/\/\*[\s\S]*?\*\//g, '');
  s = s.replace(/(^|[^:])\/\/.*$/gm, '$1');
  return s;
}
function scan(file) {
  const body = stripComments(fs.readFileSync(file, 'utf8'));
  const hits = [];
  body.split('\n').forEach((ln, i) => { if (HAN.test(ln)) hits.push({ line: i + 1, text: ln.trim() }); });
  return hits;
}

const roots = ['web/src', 'server'].map((r) => path.join(ROOT, r)).filter((r) => fs.existsSync(r));
let total = 0;
for (const root of roots) {
  for (const file of walk(root)) {
    for (const h of scan(file)) {
      total++;
      console.log(`${path.relative(ROOT, file)}:${h.line}: ${h.text}`);
    }
  }
}
if (total) {
  console.log(`\ncheck-i18n: 共 ${total} 处中文命中（目标：0）`);
  process.exit(1);
}
console.log('check-i18n: 未发现中文命中');
